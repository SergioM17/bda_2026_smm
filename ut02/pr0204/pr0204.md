```python
filtradas = []
contador = 0
 
with client.read('/datalake/renfe/raw/estaciones/estaciones_completo.csv', encoding='utf-8', delimiter='\n') as reader:
    for i, linea in enumerate(reader):
        if i == 0:
            filtradas.append(linea)
        elif "LEÓN" in linea.upper():
            filtradas.append(linea)
            contador += 1
 
client.write('/datalake/renfe/processed/estaciones_filtradas.csv', data='\n'.join(filtradas) + '\n', overwrite=True, encoding='utf-8')
print(f'Estaciones filtradas: {contador}')
```
